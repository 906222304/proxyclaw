# ============================================================
# Version: derived from `git describe --tags --always --dirty`.
# Examples:
#   Clean release tag:    v0.7.1       -> 0.7.1
#   Commits past a tag:   v0.7.1-3-gab -> 0.7.1-3-gab
#   Dirty working tree:   ...dirty     -> ...dirty
#   No git / no tag:      fallback     -> 0.7.0-unknown
# Re-run `cmake --preset ...` after tagging or switching branches to refresh.
# ============================================================
find_package(Git QUIET)
set(PROXYCLAW_VERSION_RAW "")
if(Git_FOUND AND EXISTS "${CMAKE_SOURCE_DIR}/.git")
    execute_process(
        COMMAND ${GIT_EXECUTABLE} describe --tags --always --dirty
        WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
        OUTPUT_VARIABLE PROXYCLAW_VERSION_RAW
        OUTPUT_STRIP_TRAILING_WHITESPACE
        ERROR_QUIET
    )
endif()
if(NOT PROXYCLAW_VERSION_RAW)
    set(PROXYCLAW_VERSION_RAW "${PROJECT_VERSION}-unknown")
endif()
# Strip leading "v" so "v0.7.1" becomes "0.7.1"
string(REGEX REPLACE "^v" "" PROXYCLAW_VERSION "${PROXYCLAW_VERSION_RAW}")
message(STATUS "ProxyClaw version (from git): ${PROXYCLAW_VERSION}")

# Generate C++ header and frontend TS constant.
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/generated/core")
file(WRITE "${CMAKE_BINARY_DIR}/generated/core/version.hpp"
"// Auto-generated by CMake from git describe. Do not edit.\n"
"#pragma once\n"
"#define PROXYCLAW_VERSION \"${PROXYCLAW_VERSION}\"\n"
)
file(WRITE "${CMAKE_SOURCE_DIR}/frontend/src/version.ts"
"// Auto-generated by CMake from git describe. Do not edit.\n"
"export const PROXYCLAW_VERSION = '${PROXYCLAW_VERSION}'\n"
)

# 100ms refresh-coalesce window in process_projection for batched events.
# Default ON since SYN parking: the SOCKET handler resolves unknown PIDs
# synchronously on the strand and the tree change that follows must not
# turn into an immediate full refresh + push (~35KB serialize on the
# strand) — that would hold up the next CONNECT decisions past the
# parking watchdog. User-driven changes still push immediately.
option(PROXYCLAW_PROJECTION_COALESCE "Enable projection coalesce window (100ms)" ON)

    # Parallel compilation: the Visual Studio generator parallelizes across
    # projects by default but serializes .cpp files inside a single project.
    # /MP tells cl.exe to span all cores for this target's TUs (~30 after
    # the three-layer refactor). No-op on Ninja / MSBuild batch mode.
    target_compile_options(proxyclaw PRIVATE /MP)
endif()

# WebView2 configuration
if(PROXYCLAW_ENABLE_WEBVIEW2)
    set(WEBVIEW2_PATH "${CMAKE_SOURCE_DIR}/src/ui/third_party/WebView2" CACHE PATH "Path to WebView2 SDK")

    if(EXISTS "${WEBVIEW2_PATH}/build/native/include/WebView2.h")
        message(STATUS "Found WebView2 SDK at ${WEBVIEW2_PATH}")
        include_directories("${WEBVIEW2_PATH}/build/native/include")

        if(CMAKE_SIZEOF_VOID_P EQUAL 8)
            set(WEBVIEW2_ARCH "x64")
        else()
            set(WEBVIEW2_ARCH "x86")
        endif()

        if(EXISTS "${WEBVIEW2_PATH}/build/native/${WEBVIEW2_ARCH}/WebView2LoaderStatic.lib")
            list(APPEND PROXYCLAW_LIBRARIES
                "${WEBVIEW2_PATH}/build/native/${WEBVIEW2_ARCH}/WebView2LoaderStatic.lib"
            )
            message(STATUS "Using WebView2LoaderStatic.lib")
        elseif(EXISTS "${WEBVIEW2_PATH}/build/native/${WEBVIEW2_ARCH}/WebView2Loader.dll.lib")
            list(APPEND PROXYCLAW_LIBRARIES
                "${WEBVIEW2_PATH}/build/native/${WEBVIEW2_ARCH}/WebView2Loader.dll.lib"
            )
            add_custom_command(TARGET proxyclaw POST_BUILD
                COMMAND ${CMAKE_COMMAND} -E copy_if_different
                "${WEBVIEW2_PATH}/build/native/${WEBVIEW2_ARCH}/WebView2Loader.dll"
                "$<TARGET_FILE_DIR:proxyclaw>"
            )
            message(STATUS "Using WebView2Loader.dll")
        else()
            message(WARNING "WebView2Loader library not found")
        endif()
    else()
        message(WARNING "WebView2 SDK not found at ${WEBVIEW2_PATH}")
    endif()
endif()

    # Self-contained unit tests: no driver, no admin rights, no system DNS.
    # Fake include directories go BEFORE src so they shadow production
    # logging / network headers exactly as the standalone runners do.
    function(proxyclaw_unit_test name)
        cmake_parse_arguments(T "" "" "SOURCES;FAKES;LIBS" ${ARGN})
        add_executable(${name} ${T_SOURCES})
        if(T_FAKES)
            target_include_directories(${name} BEFORE PRIVATE ${T_FAKES})
        endif()
        target_link_libraries(${name} PRIVATE ${T_LIBS})
        if(MSVC)
            target_compile_options(${name} PRIVATE /EHsc)
        endif()
        add_test(NAME ${name} COMMAND ${name}
                 WORKING_DIRECTORY "$<TARGET_FILE_DIR:${name}>")
        set_tests_properties(${name} PROPERTIES TIMEOUT 60)
    endfunction()

    proxyclaw_unit_test(api_auth_tests SOURCES tests/test_api_auth.cpp)

        SOURCES tests/test_dns_wire.cpp
